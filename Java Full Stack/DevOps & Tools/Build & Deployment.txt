------------------------------------------------------------
Create the Compose YAML File
------------------------------------------------------------

- In the projects root directory containing pom.xml and Dockerfile — create a new file named:

compose.yaml

- Add the below configuration:


```
services:
  app:
    image: quiz-app:1.0
    container_name: quiz-app-container
    ports:
      - "8079:8079"
    environment:
      SPRING_DATASOURCE_URL: jdbc:postgresql://postgres:5432/questiondb_restore
      SPRING_DATASOURCE_USERNAME: quizuser
      SPRING_DATASOURCE_PASSWORD: ${POSTGRES_PASSWORD}
    depends_on:
      postgres:
        condition: service_healthy
    restart: unless-stopped

  postgres:
    image: postgres:18
    container_name: quiz-postgres
    environment:
      POSTGRES_DB: questiondb_restore
      POSTGRES_USER: quizuser
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
    ports:
      - "127.0.0.1:5433:5432"
    volumes:
      - quiz-postgres-data:/var/lib/postgresql
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U quizuser -d questiondb_restore"]
      interval: 5s
      timeout: 3s
      retries: 10
    restart: unless-stopped

volumes:
  quiz-postgres-data:
    external: true
```

- Create .env file in the same directory and add below:

POSTGRES_PASSWORD=quiz123
